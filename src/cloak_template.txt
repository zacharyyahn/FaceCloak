#TODO:

# ---- Parameters ----

# batch_size: int batch_size for attacks. Currently only option is 1
# target_pool_size: int number of images in the target search pool
# cropped_im_size: int size of cropped image from MTCNN (square image)
# num_cloaked_images: int number of images to cloak
# num_dataset_images: float fraction of dataset to include in calculations

python src/cloak.py \
        --gallery_dataset_path "GALLERY_DATASET_PATH" \
        --probe_dataset_path "PROBE_DATASET_PATH" \
        --batch_size 1 \
        --cropped_im_size 112 \
        --num_dataset_images 1.0 \
        --norm_function "divide" \
        --cloak_save_path "SAVE_PATH" \
        --extractor_type "EXTRACTOR_TYPE" \
        --distance_function "DISTANCE" \
        --cloak_function "CF" \
        --multi_cloak_function "MULTI_CLOAK_FUNC" \
        --do_highpass "DO_HIGHPASS" \
        --do_stickers "DO_STICKERS" \
        --cloak_loss "LOSS_TYPE" \
        --multi_cloak_loss "LOSS_MULTI_TYPE" \
	--cloak_percep_loss SIM_FUNC \
        --percep_loss_weight SIM_VALUE \
        --cloak_function_iters NUM_ITERATIONS \
        --multi_cloak_function_iters MULTI_ITERATIONS \
        --cloak_function_step PERT_STEP_SIZE \
        --cloak_function_max_pert PERT_MAX \
        --multi_cloak_function_max_pert MULTI_MAX_PERT \
        --cloak_function_lr 0.5 \
        --num_gen_iterations NUM_GEN_ITERS \
        --gen_learning_rate GEN_IM_LR \
        --verbosity "error" \
        --mode "ATTACK_MODE" \
        --num_images_to_gen NITG \
        --gen_save_path "SAVE_GEN_PATH" \
        --n_to_eval "N_TO_EVAL" \
